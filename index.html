Clean Eloquent & React vs. Legacy Sprawls

          
MewCMS empowers developers with clean MVC paradigms, typed Inertia props, and reactive components.

        


        

          
          
        


        

          

            

              
              
              
            

            app/Http/Controllers/Admin/ArticleController.php
            
UTF-8 • PHP 8.3 / TSX

          

          
// MewCMS Controller (Laravel 12 + Inertia.js 2)
namespace App\Http\Controllers\Admin;

use App\Models\CmsArticle;
use Inertia\Inertia;
use Illuminate\Http\Request;

class ArticleController extends Controller
{
    public function index(Request $request)
    {
        $articles = CmsArticle::query()
            ->with(['taxonomies', 'author', 'schema'])
            ->when($request->search, fn($q, $s) => $q->where('title', 'like', "%{$s}%"))
            ->latest()
            ->paginate(15);

        // Inertia renders React 19 SPA view seamlessly without REST/GraphQL bloat
        return Inertia::render('admin/articles/index', [
            'articles' => $articles,
            'filters' => $request->only(['search', 'category'])
        ]);
    }
}
